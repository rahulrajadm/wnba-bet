# Local-only dependency for the PrizePicks DataDome workaround (pipeline/prizepicks.py).
# NOT installed on Streamlit Cloud — the deployed app (ui/app_cloud.py) never uses this
# path and runs on Underdog props only. Keep it out of requirements.txt so the cloud
# build stays lean.
#
#   pip install -r requirements-local.txt
#   playwright install chromium
#
# The workaround opens a real (headed) browser once to mint a DataDome cookie, then reuses
# it via curl_cffi, so a desktop session (not a headless server) is required.
#
# Note: curl_cffi itself is a separate, unrelated dependency of requirements.txt too —
# pipeline/cloud_data.py (ESPN fetch) uses its browser-impersonation mode to get past
# Akamai bot detection, no cookie/browser session needed there. It's listed here again
# only because playwright still requires a desktop session.
playwright>=1.40.0
